    function renderTransactions() {
      const container = document.getElementById('tx-list-container');
      container.innerHTML = '';
      appState.transactions.forEach(tx => {
        const isPos = tx.amount > 0;
        const formatted = isPos ? '+' + formatUSD(tx.amount) : formatUSD(tx.amount);
        const statusClass = (tx.status || 'Completed').toLowerCase() === 'pending' ? 'pending' : 'completed';

        const btn = document.createElement('button');
        btn.type = 'button';
        btn.className = 'tx-row';
        btn.onclick = () => openReceiptModal(tx);
        btn.innerHTML = `
          <div class="tx-info">
            <div class="tx-title">${escapeHtml(tx.displayTitle || tx.fullTitle)}</div>
            <div class="tx-date">${escapeHtml(tx.date)}</div>

          <div class="tx-amount ${isPos ? 'positive' : ''}">${formatted}</div>
          <span class="tx-status-pill ${statusClass}">${escapeHtml(tx.status || 'Completed')}</span>
        `;
        container.appendChild(btn);
      });
    }

    function escapeHtml(str) {
      return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;');
    }

    function copyPayTag() {
      const tag = appState.payTag;
      if (navigator.clipboard && window.isSecureContext) {
        navigator.clipboard.writeText(tag).catch(() => fallbackCopyText(tag));
      } else {
        fallbackCopyText(tag);
      }
      showToast(`Copied ${tag}`);
    }

    function fallbackCopyText(text) {
      const ta = document.createElement('textarea');
      ta.value = text;
      document.body.appendChild(ta);
      ta.select();
      document.execCommand('copy');
      document.body.removeChild(ta);
    }

    function submitSendMoney(e) {
      e.preventDefault();
      let tag = document.getElementById('inp-send-tag').value.trim();
      if (!tag.startsWith('@')) tag = '@' + tag;
      const amt = parseFloat(document.getElementById('inp-send-amt').value);
      const note = document.getElementById('inp-send-note').value.trim() || 'Instant @PayTag transfer';

      if (isNaN(amt) || amt <= 0) return;
      if (amt > appState.balance) {
        showToast('Insufficient balance', true);
        return;
      }

      appState.balance -= amt;
      const shortTitle = `Sent to ${tag.length > 5 ? tag.slice(0, 4) + '...' : tag}`;
      appState.transactions.unshift({
        id: 'TX-' + Math.floor(1000 + Math.random() * 9000),
        displayTitle: shortTitle,
        fullTitle: `Sent to ${tag}`,
        recipient: tag,
        date: 'Nov 26, 2026',
        amount: -amt,
        status: 'Completed',
        note
      });

      saveState();
      renderAll();
      closeModal();
      showToast(`Sent ${formatUSD(amt)} to ${tag}`);
    }

    function submitAddMoney(e) {
      e.preventDefault();
      const amt = parseFloat(document.getElementById('inp-add-amt').value);

        status: 'Completed',

      renderAll();
      closeModal();
      showToast(`Added ${formatUSD(amt)} to balance`);
    }

    function submitWithdraw(e) {

    function applyAdminAccountChanges(e) {
      e.preventDefault();
      const newBal = parseFloat(document.getElementById('adm-balance').value);
      let newTag = document.getElementById('adm-paytag').value.trim();
      if (newTag && !newTag.startsWith('@')) newTag = '@' + newTag;
      const newName = document.getElementById('adm-name').value.trim();
      const verified = document.getElementById('adm-verified').checked;

      if (!isNaN(newBal)) appState.balance = newBal;
      if (newTag) appState.payTag = newTag;
      if (newName) appState.displayName = newName;
      appState.isVerified = verified;

      saveState();
      renderAll();
      closeModal();
      showToast('Admin changes saved');
    }

    function injectAdminTransaction(e) {
      e.preventDefault();
      const dir = document.getElementById('adm-tx-dir').value;
      const status = document.getElementById('adm-tx-status').value;
      const label = document.getElementById('adm-tx-label').value.trim();
      const rawAmt = parseFloat(document.getElementById('adm-tx-amt').value);
      const dateStr = document.getElementById('adm-tx-date').value.trim() || 'Nov 26, 2026';

      if (isNaN(rawAmt) || rawAmt <= 0 || !label) return;

      const signedAmt = dir === 'out' ? -Math.abs(rawAmt) : Math.abs(rawAmt);
      const shortTitle = label.length > 12 ? label.slice(0, 10) + '...' : label;

      appState.transactions.unshift({
        id: 'TX-' + Math.floor(1000 + Math.random() * 9000),
        displayTitle: shortTitle,
        fullTitle: label,
        recipient: label.includes('@') ? '@' + label.split('@')[1].split(' ')[0] : appState.payTag,
        date: dateStr,
        amount: signedAmt,
        status,
        note: 'Admin ledger entry'
      });

      saveState();
      renderAll();
      closeModal();
      showToast('Custom transaction added');
    }

    function resetAllDemoData() {
      localStorage.removeItem(STORAGE_KEY);
      appState = JSON.parse(JSON.stringify(DEFAULT_STATE));
      saveState();
      renderAll();
      closeModal();
      showToast('Demo data restored to defaults');
    }

    /* ========================================================================
       SEASONAL DECORATIVE AUTUMN LEAVES & PUMPKINS
       ======================================================================== */
    function renderSeasonParticles() {
      const layer = document.getElementById('leaves-layer');
      layer.innerHTML = '';
      if (!appState.showSeasonParticles) return;

      const icons = ['🍂', '🍁', '🎃', '🍃'];
      for (let i = 0; i < 10; i++) {
        const el = document.createElement('span');
        el.className = 'season-particle';
        el.textContent = icons[i % icons.length];
        el.style.left = `${(i * 10 + 4) % 95}vw`;
        el.style.fontSize = i % 4 === 2 ? '13px' : '15px';
        el.style.animationDuration = `${11 + (i % 4) * 2.4}s`;
        el.style.animationDelay = `${-(i * 1.7)}s`;
        layer.appendChild(el);
      }
    }
